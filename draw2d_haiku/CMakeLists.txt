cmake_minimum_required(VERSION 3.17)
project(draw2d_haiku CXX)
add_library(${PROJECT_NAME} SHARED backend.cpp _factory.cpp native.cpp)
add_library(static_${PROJECT_NAME} STATIC backend.cpp _factory.cpp native.cpp)
foreach(target ${PROJECT_NAME} static_${PROJECT_NAME})
 target_compile_features(${target} PRIVATE cxx_std_20)
 target_compile_options(${target} PRIVATE -fpermissive)
 target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_definitions(${target} PRIVATE _draw2d_haiku_project)
endforeach()
target_compile_definitions(static_${PROJECT_NAME} PRIVATE CUBE)
target_link_libraries(${PROJECT_NAME} PRIVATE acme apex aqua aura be)
target_link_options(${PROJECT_NAME} PRIVATE -Wl,-z,defs)
target_link_libraries(static_${PROJECT_NAME} PRIVATE static_acme static_apex static_aqua static_aura be)
add_executable(draw2d_haiku_native_test EXCLUDE_FROM_ALL native_test.cpp native.cpp)
target_link_libraries(draw2d_haiku_native_test PRIVATE be)
target_compile_features(draw2d_haiku_native_test PRIVATE cxx_std_20)
